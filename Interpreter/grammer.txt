How To Read Grammer:
"<value>" = character(s) expected.
-> = something consists of this.
() = grouping.
| = Or.
* = 0 or more times.
? = 0 or 1 time.
; = end.
IDENTIFIER = variable name token.
EOF = end-of-file token.

Grammer:
program -> statement* EOF ;


statement -> variableDeclaration
            | ifStatement
            | loopStatement
            | printStatement 
            | expressionStatement 
            | blockStatement ;

blockStatement -> "{" statement* "}" ;

variableDeclaration -> 
        "var" IDENTIFIER 
        "->" ( "number" | "string" | "bool" ) // | IDENTIFIER in future for obj types ect 
        "=" expression ";" ;

ifStatement -> "if" "(" expression ")" statement
            ( "else" statement )? ;

loopStatement -> "loop"
            ( "(" expression ")" )? 
            statement ;

printStatement -> ( "out" | "outln" ) expression ";" ;

expressionStatement -> expression ";" ;

array -> "[" expression ( "," expression )* "]" | "[" "]" ;

expression -> assignment ;

assignment -> IDENTIFIER "=" assignment | equality ;

logicOr -> logic_and ( "or" logic_and )* ;

logicAnd -> equality ( "and" equality )* ;

equality -> comparison
        ( ("==" | "!=") comparison )* ;

comparison -> term
        ( (">" | ">=" | "<" | "<=") term )* ;

term -> factor
       ( ("+" | "-") factor )* ;

factor -> unary
         ( ("*" | "/") unary )* ;

unary -> ("!" | "-") unary
      | "typeof" unary
      | primary ;

type ->   "number"
        | "string"
        | "bool" ;

arrayType -> type "[]" ;

primary -> NUMBER
        | STRING
        | TRUE
        | FALSE
        | IDENTIFIER
        | "(" expression ")"
        | primary "[" expression "]" ;
